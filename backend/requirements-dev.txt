# pytest — unit test runner.
# PMP §21 Testing Strategy, "Unit Tests".
pytest==9.1.1

# pytest-asyncio — async test support.
# Required by the async endpoints in Phase 1 (lifespan, /health/ready)
# and mandatory later for the SSE query endpoint (PMP §14).
# Requires pytest>=8.4,<10; satisfied by pytest 9.1.1 above.
pytest-asyncio==1.4.0

# httpx — HTTP client, and the ASGI transport behind
# fastapi.testclient.TestClient.
# PMP §15 Operations endpoints (/health, /health/ready) are the first
# things tested. NOTE: starlette's TestClient is sensitive to httpx
# major versions; the 0.28.x line is the one resolved against
# starlette 1.7.0 in this set.
httpx==0.28.1

# ruff — linter and formatter, the lint gate for CI.
# PMP §23 technology stack (CI/CD: GitHub Actions). Distributed as a
# standalone Rust binary; contributes no Python dependencies.
ruff==0.16.9

# mypy — static type checker.
# PMP §8 Maintainability, "Type hints". Adds ast-serialize, librt,
# mypy_extensions, pathspec and typing-extensions as transitive deps.
mypy==2.3.1
